\section{Research problem, research backgound, and our main contribution}
\begin{frame}{Numerical methods for a linear system $AB \bm{x} = \bm{b}$}
	We consider numerically solving a linear system in the form  
	\begin{equation}\label{eq:fls}
	AB \bm{x} = \bm{b}
	.
	\end{equation}	
 

In our discussion, a numerical method for a mathematical problem is:
\begin{itemize}
	\item applicable in finite precision arithmetics
	\item terminates in finite steps
	\item error of the approximated solution is clear 
\end{itemize}
\textbf{Research subject}
In the present research, we study two numerical approaches, \textit{the separate approach} and \textit{the combined approach}.
\end{frame}

\begin{frame}{Research Background}
	% \vspace{0pt}
	Consider the three numerical methods for solving a linear system $A \bm{x} = \bm{b}$
	\begin{enumerate}
		\item Direct methods
		\item Stationary iterative methods
		\item Krylov subspace methods
	\end{enumerate}
	All of these methods can be corperated to solve linear system $AB \bm{x} = \bm{b}$.
		\begin{table}[t]
		\begin{tabular}{l c c}
			\toprule
			 & Separate & Combined\\	
			\midrule
			Direct&  \maru & \basu \\
			Stationary & \maru & \basu \\
			Krylov & \maru & \maru \\
			\bottomrule
		\end{tabular}
		\caption{Compatibility with two approaches. \maru indicates compatible, \basu uncompatible}
		\end{table}
\end{frame}

\begin{frame}{Our main contribution}
	\textbf{Main goal:} To understand the numerical behavior of the two approaches\\
	\textbf{Our findings so far:} In particular, we show that.
		\begin{itemize}
			\item accuracy loss of \textit{the separate approach} when stopping criteria is set trivially
			\item better performance of \textit{the combined approach} when the condition number $\kappa(AB)$ is good
		\end{itemize}
		\textbf{Current result:} As a first theoretical step toward understanding this behavior, we establish theorem on the relative residual bound of \textit{the separate approach}
\end{frame}

\section{Numerical approaches and a theorem on the relative residual (preliminary)}
\begin{frame}{Approach 1: The separate approach (Two-stage solve)}
	Approach 1 is based on the equivalent expression
	\begin{equation}
	\begin{cases}
		A \bm{y} = \bm{b},\\
		B \bm{x} = \bm{y}.
	\end{cases}
	\end{equation}
Two linear systems are sequentially solved in the following pipeline
\begin{center}
solve $A \bm{y} = \bm{b}$ $|$ solve $B \bm{x} = \bm{y}$
\end{center}
\begin{block}{Example: on its numerical realization}
	\vspace{1pt}
	\begin{tabular}{c c}		
direct $A \bm{y} = \bm{b}$ $|$ direct $B \bm{x} = \bm{y}$ & bicgstab $A \bm{y} = \bm{b}$ $|$ bicgstab $B \bm{x} = \bm{y}$\\
stationary $A \bm{y} = \bm{b}$ $|$ stationary $B \bm{x} = \bm{y}$ & direct $A \bm{y} = \bm{b}$ $|$ bicgstab $B \bm{x} = \bm{y}$\\
krylov $A \bm{y} = \bm{b}$ $|$ krylov $B \bm{x} = \bm{y}$ & gmres $A \bm{y} = \bm{b}$ $|$ bicgstab $B \bm{x} = \bm{y}$
	\end{tabular}
\end{block}
\end{frame}

\begin{frame}{Approach 2: The combined approach (Direct solve)}
	Approach 2 solves the linear system $ AB \bm{x} = \bm{b}$ by the form $C \bm{x} = \bm{b}, C = AB$ 
	\begin{block}{Different realizations of $C=AB$, when $A$, $B$ sparse}
		\vspace{1pt}
	% When applying approach 1 to reconstruct the solution of $AB \bm{x} = \bm{b}$, at least one of the following forms will be exploited:
		$A \in \mathbb{R} ^{n \times n}, B \in \mathbb{R} ^{n \times n}, N_{nz}(A)=O(n), N_{nz}(B) = O(n)$
	\begin{itemize}
		\item \textit{\sout{Explicit formulation}}: matrix multiplication $AB$ $\rightarrow O({n}^{3})$
		\item \textit{Implicit formulation}: matrix-vector product $\underbracket{A(B \bm{b})}_{\text{Krylov}} \rightarrow O(n)$
	\end{itemize}
\end{block}
\begin{block}{Example: on its memory usage, accuracy, and convergence rate}
	\vspace{0pt}
	Memory usage \textit{(implicit formulation)}: $mO(n)$, $m$ is number of iterations\\
	Accuracy \textit{(Krylov subspace methods)}: relative residual\\
	Convergence rate \textit{(Krylov subspace methods)}: the condition number $\kappa(AB)$
\end{block}
% The direct combined approach is worth considering (in computational cost) when \textcolor{red}{Krylov subspace method} is applied.
\end{frame}

\begin{frame}{\textit{Separate}: our theorem on the relative residual bound}
	\begin{thm}[Relative residual of the separate approach]\label{relthm}
		\vspace{0pt}
		Consider matrices $A , B , C \in \mathbb{C}^{n \times n}$, vectors $\bm{y}, \bm{b}, \bm{x}, \bm{r}_{A}, \bm{r}_{B} \in \mathbb{C}^{n}$ ($C = AB$, $\bm{b} \neq \bm{0}$), of which the following relations are satisfied:
		\begin{align*}
			A\bm{y} & = \bm{b} + \bm{r}_{A}, \quad (\varepsilon_{A} = \frac{\|\bm{r}_{A}\|}{\|\bm{b}\|}) \\
			B \bm{x} & = \bm{y} + \bm{r}_{B}, \quad (\varepsilon_{A} = \frac{\|\bm{r}_{A}\|}{\|\bm{b}\|})
		\end{align*}
	then we have the following inequatlity:
	\begin{equation}
	\frac{\|\bm{b} - AB \bm{x}\|}{\|\bm{b}\|} \leq \kappa(A) \varepsilon_{B} ( 1 + \varepsilon_{A} ) + \varepsilon_{A}	
	.
	\end{equation}
	\vspace{-2em}
	\begin{proof}
		See Appendices	
	\end{proof}	
	\end{thm}
\end{frame}

			\item \textit{the combined approach:} specific to Krylov methods, condition of $AB$
			\item accuracy of the two approaches: be careful of \textit{separate} since the relative residual is not well bounded
		\end{enumerate}
	\end{block}

		\item[Q2:] (Comparison of approaches) when \textit{the combined approach} is a good choice \& those when \textit{the separate approach} is a good choice
	\end{enumerate}
	\end{block}
\end{frame}

\section{Our findings on the model problem (Toeplitz matrices)}
\begin{frame}{Before the discussion: how to set up stopping criteria}
	Solve the linear system $AB \bm{x} = \bm{b}$ within the tolerance $\varepsilon_{tol}$, s.t.
	\begin{equation}\label{toldef}	
		\text{relative residual \textit{(rel. res.)}} = \cfrac{\|\bm{b} - A B \bm{x}\|}{\|\bm{b}\|} \leq \varepsilon_{tol}
	.
	\end{equation}
	\sout{However}, our setting:

	\begin{block}{The combined method}
		\begin{equation*}
			\underbracket{\text{Krylov}}_{\varepsilon_{c} = \varepsilon_{tol}} \text{ } AB \bm{x} = \bm{b}\xrightarrow{\text{leads to}} \textit{rel. res.} \leq \varepsilon_{tol}
		\end{equation*}		
	\end{block}
	\begin{block}{The separate method}
		\begin{equation*}
			\underbracket{\text{Krylov}}_{\epsilon_1 = \varepsilon_{tol}} \text{ } A \bm{y} = \bm{b} \text{ }| \text{ } \underbracket{\text{Krylov}}_{\varepsilon_2 = \varepsilon_{tol}} \text{ } B \bm{x} = \bm{y}\xrightarrow{\alert{\text{leads to??}}}  \hbox{\sout{$\textit{rel. res.} \leq \varepsilon_{tol}$}}
		\end{equation*}	
	\end{block}
\end{frame}

\begin{frame}{Discussions: toward a possible proper stopping criteria (open)}
	\begin{col}[Relations to \textit{the separate approach}]
		\vspace{0pt}
		Consider our separate approach:	Solve $A \bm{y} = \bm{b}$ $|$ Solve $B \bm{x} = \bm{y}$. Let $\bm{y}$ be the output after solving $A \bm{y} = \bm{b}$, and $\bm{x}$ be the output after solving $B \bm{x} = \bm{y}$. Residuals $\bm{r}_{A}, \bm{r}_{B}$ represent the inaccuracy of the approximated solution $\bm{y}, \bm{x}$. $\varepsilon_{A}, \varepsilon_{B}$ are the true relative residuals.
		\begin{align*}
			A\bm{y} & = \bm{b} + \bm{r}_{A}, \quad (\varepsilon_{A} = \frac{\|\bm{r}_{A}\|}{\|\bm{b}\|}) \\
			B \bm{x} & = \bm{y} + \bm{r}_{B}, \quad (\varepsilon_{A} = \frac{\|\bm{r}_{A}\|}{\|\bm{b}\|})
		\end{align*}
		\alert{Setting up the stopping criteria $\varepsilon_{1}, \varepsilon_{2}$ s. t. the relative residuals satisfies} $\kappa(A) \varepsilon_{B} ( 1 + \varepsilon_{A} ) + \varepsilon_{A} \leq \varepsilon_{tol}$ \textit{(Note that in Theorem \ref{relthm} $\frac{\|\bm{b} - AB \bm{x}\|}{\|\bm{b}\|} \leq \kappa(A) \varepsilon_{B} ( 1 + \varepsilon_{A} ) + \varepsilon_{A}$)}\\
		\alert{Then, $\bm{x}$ is a good approximation whose relative reisudal is bounded.}
	\end{col}
\end{frame}


\section{Conclusions and future work}
\begin{frame}{Conclusions}
	\begin{enumerate}
		\item[A1] \textit{(Separate)} Setting up solvers for the two stages is generally complicated. However, a theoretical analysis on the relative resiual (of which Krylov subspace methods take good control) is delivered, by which practical stopping criteria become possible.
		\item[A2] \textit{(Combined)} Although not a thorough analysis, we found a $\mathrm{cond}(AB)$ is a great factor which influences the performance of \textit{the combined approach}.
	\end{enumerate}	

\end{frame}
\begin{frame}{Future work}
	\begin{itemize}
		\item After developing practical stopping criteria, the two approaches could be compared within larger range of problems
		\item Could the preconditiond linear solvers be analysed with the same tools?
	\end{itemize}
\end{frame}

\begin{frame}{A residual norm analysis}
	Considering the two-stage method:
	\begin{equation}
		\|r\| = \|b - ABx\| = \|Ay - A(Bx)\| = \|A(y - Bx)\| \le \|A\| \|y - Bx\|
	\end{equation}
	This shows that even if the second system $Bx=y$ is solved with a small residual, the global residual can be magnified by $\|A\|$.
\end{frame}
