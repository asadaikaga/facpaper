\section{Research problem}
\begin{frame}{Research problem}
	We consider numerically solving the linear systems in the form  
	\begin{equation}\label{eq:fls}
	AB \bm{x} = \bm{b}
	.
\end{equation}	
 

A Numerical method for a mathematical problem should be:
\begin{itemize}
	\item Applicable in finite precision arithmetics
	\item Terminates in finite steps
	\item Error of the approximated solution is clear 
\end{itemize}
\textbf{Research Objective}
In the present research, numerical methods for problem \eqref{eq:fls} are studied under different conditions of the given problem. 
\end{frame}

\begin{frame}{Numerical methods for a linear system}
	Two solve a linear system $A \bm{x} =\bm{b}$, the following three numerically are frequently considered:
	\begin{enumerate}
		\item Direct methods
		\item Stationary iterative methods
		\item Krylov subspace methods
	\end{enumerate}
	However, when corperate these methods to solve linear system $AB \bm{x} = \bm{b}$, Krylov subspce methods are recommended:
	\begin{itemize}
		\item computational complexity of  $A(B\bm{b})$: $O(n)$, when $A,B$ are sparse \textcolor{red}{ (Krylov iterations)}
		\item complexity of $AB$: $O(n^3)$ \textcolor{red}{ (Other two methods)}
	\end{itemize}
\end{frame}

\begin{frame}{Approach 1: Direct combined approach}
	Approach 1 solve the linear system $ AB \bm{x} = \bm{b}$ by the form $C \bm{x} = \bm{b}, C = AB$ 
	\begin{block}{Different realizations of $C=AB$}
	When applying approach 1 to reconstruct the solution of $AB \bm{x} = \bm{b}$, at least one of the following forms will be exploited:
	\begin{itemize}
		\item Explicit formulation: matrix multiplication $AB$
		\item Implicit formulation: matrix-vector product  $A(B \bm{b})$
	\end{itemize}
\end{block}
The direct combined approach is worth considering (in computational cost) when \textcolor{red}{Krylov subspace method} is applied.
\end{frame}
\begin{frame}{Approach 2: Two-stage approach}
	Approach 2 is based on the equivalent expression
	\begin{equation}
	\begin{cases}
		A \bm{y}& = \bm{b}\\
		B \bm{x} & = \bm{y}
	\end{cases}
	.
	\end{equation}
Two linear systems are sequentially solved in the following pipeline
\begin{center}
solve $A \bm{y} = \bm{b}$ $|$ solve $B \bm{x} = \bm{y}$
\end{center}
\begin{block}{On numerical realization of this approach}

Unlike approach 1, all methods (direct methods, stationary iterative methods, Krylov subspace method) can be corporated with the two-stage approach
\end{block}
\end{frame}
\begin{frame}{Numerical experiments}
	\textbf{Experiment objective}:	
	\begin{itemize}
		\item Compare the convergence speed of Approach 1 and Approach 2.
		\item Analyze the impact of condition numbers $\kappa(A)$ and $\kappa(B)$ on the final accuracy.
		\item Evaluate the memory efficiency of implicit vs explicit formulations.
	\end{itemize}
\end{frame}
\begin{frame}{Algorithmic behaviour in relation to the condition number (residual norm)}
	\begin{block}{Approach 1: Direct Combined}
		The residual norm $\|b - ABx\|$ is minimized directly by the Krylov solver. Convergence rate is primarily determined by $\kappa(AB)$.
	\end{block}
	\begin{block}{Approach 2: Two-Stage}
		The global residual depends on the accuracy of both stages. 
		If $Ay=b$ is solved with residual $r_1$ and $Bx=y$ with $r_2$, the final residual is $r = b - ABx = r_1 - Ar_2$.
	\end{block}
\end{frame}
\begin{frame}{Algorithmic behaviour in relation to the condition number (error norm)}
	\begin{itemize}
		\item \textbf{Direct Approach}: Error is bounded by the condition number of the product $C = AB$.
		\item \textbf{Two-Stage Approach}: Error in $y$ ($\| \delta y \| \propto \kappa(A)$) propagates to $x$.
		\item \textbf{Observation}: If $A$ is extremely ill-conditioned, the intermediate solution $y$ may lose significant precision, leading to stagnation in the second stage.
	\end{itemize}
\end{frame}
\begin{frame}{Summary and future directions}
	\begin{itemize}
		\item \textbf{Summary}: Direct approach is more robust against ill-conditioning but computationally intensive if sparsity is not preserved. Two-stage approach is flexible but requires careful error management.
		\item \textbf{Future Work}: 
		\begin{itemize}
			\item Developing adaptive switching between approaches based on $\kappa(A), \kappa(B)$.
			\item Extending the analysis to more than two factors ($ABCx=b$).
		\end{itemize}
	\end{itemize}
\end{frame}
\begin{frame}{Research plans}
	\begin{enumerate}
		\item Formalize the theoretical error bounds for the two-stage approach.
		\item Implement test suites using high-dimensional sparse matrices from the SuiteSparse collection.
		\item Investigate the performance of different Krylov variants (GMRES, BiCGSTAB) on these structures.
	\end{enumerate}
\end{frame}
\begin{frame}{Special cases}
	\textbf{Preconditioning}:
	\begin{itemize}
		\item Should we precondition $A$ and $B$ separately or the product $AB$?
		\item Left preconditioning: $M^{-1}ABx = M^{-1}b$.
		\item Split preconditioning: $M_A^{-1}AM_B^{-1}BM_B x = M_A^{-1}b$.
	\end{itemize}
\end{frame}

\begin{frame}{A residual norm analysis}
	Considering the two-stage method:
	\begin{equation}
		\|r\| = \|b - ABx\| = \|Ay - A(Bx)\| = \|A(y - Bx)\| \le \|A\| \|y - Bx\|
	\end{equation}
	This shows that even if the second system $Bx=y$ is solved with a small residual, the global residual can be magnified by $\|A\|$.
\end{frame}
